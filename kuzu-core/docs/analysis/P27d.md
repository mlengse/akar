    row_idx: usize,

    }

impl PartialOrd for HeapItem { fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> { Some(self.cmp(other)) } }
impl Eq for HeapItem {}
impl PartialEq for HeapItem { fn eq(&self, other: &Self) -> bool { self.cmp(other) == std::cmp::Ordering::Equal } }

fn k_way_merge(...) -> Vec<usize> {
    let mut heap: BinaryHeap<HeapItem> = BinaryHeap::with_capacity(blocks.len());
    for bi in 0..blocks.len() {
        if !blocks[bi].is_empty() { heap.push(HeapItem { block_idx: bi, row_idx: 0 }); }

    let mut result = Vec::with_capacity(total_rows);
    while let Some(item) = heap.pop() {
        result.push(blocks[item.block_idx][item.row_idx]);
        let next = item.row_idx + 1;
        if next < blocks[item.block_idx].len() {
            heap.push(HeapItem { block_idx: item.block_idx, row_idx: next });
        }
    }
    result
}
